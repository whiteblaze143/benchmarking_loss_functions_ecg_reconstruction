11.3 Ten-to-fifteen-epoch learning trajectories

Code 
convergence["configuration"] = (convergence.run_name
    .str.replace(r"^conv(?:10|15)e_", "", regex=True)
    .str.replace(r"_s\d+_l[01]$", "", regex=True))
trajectory_metrics = ["val_missing_pearson", "val_missing_pearson_p05",
                      "miou_wave", "boundary_f1_smoke", "P_iou", "QRS_iou", "T_iou"]
trajectory = convergence[convergence.expected_epochs.isin([10, 15])].melt(
    id_vars=["configuration", "observed_lead", "expected_epochs", "run_name"],
    value_vars=trajectory_metrics, var_name="metric", value_name="value").dropna()
matched_ids = (trajectory.groupby(["configuration", "observed_lead", "metric"])
               .expected_epochs.nunique().loc[lambda x: x == 2].index)
trajectory = trajectory.set_index(["configuration", "observed_lead", "metric"])
trajectory = trajectory.loc[trajectory.index.isin(matched_ids)].reset_index()

fig_traj = px.line(
    trajectory, x="expected_epochs", y="value", color="configuration",
    line_dash="observed_lead", facet_col="metric", facet_col_wrap=2,
    markers=True, hover_name="run_name",
    title="Matched 10-to-15 Epoch Learning Trajectories"
)
fig_traj.update_xaxes(dtick=5)

internal_matrix = spatial_completed.pivot_table(
    index=["factorial_mask", "observed_lead"], columns="variant",
    values="best_val_missing_pearson")

    index=["factorial_mask", "input_lead"], columns="variant",

transport_3d_rows = []
for mask in sorted(internal_matrix.index.get_level_values(0).unique()):
    for lead_name, lead_index in [("I", 0), ("II", 1)]:
        internal_delta = (internal_matrix.loc[(mask, lead_name), "e1_panorama_film"] -
                          internal_matrix.loc[(mask, lead_name), "pg1_permuted_geometry"])
        ext_f1_delta = (external_matrix.loc[(mask, lead_index), ("primary_mean_micro_f1_20ms", "e1_panorama_film")] -
                        external_matrix.loc[(mask, lead_index), ("primary_mean_micro_f1_20ms", "pg1_permuted_geometry")])
        ext_p05_delta = (external_matrix.loc[(mask, lead_index), ("primary_signal_pearson_p05", "e1_panorama_film")] -
                         external_matrix.loc[(mask, lead_index), ("primary_signal_pearson_p05", "pg1_permuted_geometry")])
        transport_3d_rows.append({
            "mask": mask, "lead": lead_name,
