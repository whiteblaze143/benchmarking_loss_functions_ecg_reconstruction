import pandas as pd

EVAL_DB = source_path("results/semiseg_ludb/semiseg_ludb.sqlite")
TRAIN_DB = source_path("results/semiseg_ludb_training/semiseg_ludb_mt_evaluation.sqlite")

def semiseg_tables(path, source):
    runs = read_sql(path, "select * from runs")
    if not runs.empty:
        runs = pd.concat([runs.drop(columns="summary_json"),
                          json_normalize(runs.summary_json, "metric.")], axis=1)
        runs["source"] = source
    streams = read_sql(path, "select * from stream_metrics")
    boundaries = read_sql(path, "select * from boundary_metrics")
    if not streams.empty: streams["source"] = source
    if not boundaries.empty: boundaries["source"] = source
    return runs, streams, boundaries

er, es, eb = semiseg_tables(EVAL_DB, "evaluation package")
tr, ts, tb = semiseg_tables(TRAIN_DB, "mean-teacher package")
runs = pd.concat([er, tr], ignore_index=True)
streams = pd.concat([es, ts], ignore_index=True)
boundaries = pd.concat([eb, tb], ignore_index=True)

10.3 Boundary event accounting

event = (boundaries.groupby(["source", "run_id", "boundary"])
         .agg(reference_events=("reference_events", "sum"),
              predicted_events=("predicted_events", "sum"),
              true_positive=("true_positive", "sum"),
              false_positive=("false_positive", "sum"),
              false_negative=("false_negative", "sum"))

def semiseg_tables(path, source):
    runs = read_sql(path, "select * from runs")
    if not runs.empty:
        runs = pd.concat([runs.drop(columns="summary_json"),
                          json_normalize(runs.summary_json, "metric.")], axis=1)
        runs["source"] = source
    streams = read_sql(path, "select * from stream_metrics")
    boundaries = read_sql(path, "select * from boundary_metrics")
    if not streams.empty: streams["source"] = source
    if not boundaries.empty: boundaries["source"] = source
    return runs, streams, boundaries
